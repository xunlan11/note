\subsection[数学模型]{数学模型}
%------------------------------------------------
\subsubsection[拉氏变换]{\textcolor{red}{拉氏（Laplace）变换}\tip{拉氏变换（性质、常用变换）}}
%------------------------------------------------
\paragraph{变换}
\textcolor{red}{单边衰减的傅里叶变换}，记$ s = \sigma + j \omega t, \quad F(s) = L[f(t)] = \int_{\textcolor{red}{0}}^{\infty} f(t) e^{-st} dt = \int_0^{\infty} f(t) e^{-\sigma t} e^{-j \omega t} dt $。而傅里叶变换为$ F(\omega) = \int_{-\infty}^{\infty} g(t) e^{-j \omega t} dt $。
%------------------------------------------------
\paragraph{\textcolor{red}{性质}}
\begin{itemize}
\item \textcolor{red}{微分定理}（非零初始条件）：$ L[f^{(n)}(t)] = s^n F(s) - f^{(n - 1)}(0) - \dots - s^{n - 1} f(0) $。
\item 线性：$ f(t) = f_1(t) + f_2(t) \Rightarrow F(s) = F_1(s) + F_2(s) $。
\item
\begin{itemize}
\item \textcolor{red}{终值定理}：$ f(\infty) = \lim_{s \to 0} \textcolor{red}{s} F(s) $。
\item 初值定理：$ f(0^+) = \lim_{t \to 0^+} f(t)  = \lim_{s \to \infty} sF(s) $。
\end{itemize}
\item 位移定理：$ L[f(t)] = F(s) \Rightarrow L[f(t - \tau)] = e^{-\tau s}F(s), \quad L[f(t)e^{\alpha t}] = F(s - \alpha) $。
\item 相似定理：$ L[f(t)] = F(s) \Rightarrow L[f(\frac{t}{a})] = a F(as) $。
\end{itemize}

\paragraph{应用}
微分方程（时域）$ \overset{\text{拉氏变换}}{\Longrightarrow} $代数方程（s域）$ \overset{\text{求解}}{\Longrightarrow} $输出量表达式（s域）$ \overset{\text{反拉氏变换}}{\Longrightarrow} $方程解（时域）。

\subsubsection[控制微分方程]{控制微分方程}
输入输出方程$ \Rightarrow $按信号传递顺序列微分方程$ \Rightarrow $消去中间变量$ \Rightarrow $归一为标准形式。
%------------------------------------------------
\paragraph{\textcolor{red}{非线性的线性化}}
\begin{itemize}
\item 非线性特性不强时可忽略。
\item 偏差法：在平衡点（状态不再发生变化）附近线性化，使用泰勒展开$ y = f(x) = f(x_0) + \frac{d f(x)}{dx}|_{x_0} \cdot \Delta x + o(\Delta x) $，得到增量线性化方程：$ \Delta y = \frac{d f(x)}{dx}|_{x_0} \cdot \Delta x $。
\end{itemize}
%------------------------------------------------
\paragraph{解与模态}
\begin{itemize}
\item 解 = 特解 + 通解（特征根，代表自由运动）。
\item 运动模态（振型）：
\begin{itemize}
\item 单根$ \lambda $：$ e^{\lambda t} $。
\item 重根$ \lambda^m $：$ t^i e^{\lambda t}, \quad i = 0, 1, 2, \dots, m - 1 $。
\item 共轭复根$ \sigma \pm j \omega $：$ e^{(\sigma \pm j \omega)t} = e^{\sigma t} \sin \omega t + e^{\sigma t} \cos \omega t $。
\end{itemize}
\end{itemize}

\subsubsection[传递函数]{\textcolor{red}{传递函数}\tip{传递函数}}
%------------------------------------------------
\paragraph{概念}
\begin{itemize}
\item 定义：系统在\textcolor{red}{零初始条件}下，输出、输入拉式变换之比。
\item 性质：反应线性系统特性，决定于系统结构和参数，与输入量无关，对应微分方程为$ s $的有理真分式。
\item 单位脉冲响应（等效反馈函数）$ g(t) $：$ \textcolor{red}{G(s) = L[g(t)]} $。
\item 输入输出关系：线性定常系统中，若输入是微积分关系，则输出也是，常数用零初始条件确定。
$$ \textcolor{red}{r_2(t) = \dot{r_1(t)} \Rightarrow c_2(t) = \dot{c_1(t)}} $$
\end{itemize}
%------------------------------------------------
\paragraph{\textcolor{red}{响应计算}}
\begin{itemize}
\item 零初始条件：$ \textcolor{red}{ R(s) = L[r(t)] \Rightarrow C(s) = G(s)R(s) \Rightarrow c(t) = L^{-1}[C(s)]} $。
\item 非零初始条件：由传递函数反推微分方程，使用微分定理进行拉氏变换。
\end{itemize}

分式有理化计算：
\begin{itemize}
\item 待定系数法。
\item 留数法：
\begin{itemize}
\item 单根：$ \frac{A}{s - a} \Rightarrow A = \lim_{s \to a} C(s) (s - a) $。
\item $ m $重根：$ \frac{A_{m - j}}{(s - a)^{m - j}} \Rightarrow A_{m - j} = \frac{1}{j!} \lim_{s \to a} \frac{d^j[C(s) (s - a)^m]}{d s^j} $。
\end{itemize}
\end{itemize}
%------------------------------------------------
\paragraph{\textcolor{red}{零极点}}
$$ G(s) = \frac{C(s)}{R(s)} = \frac{b_0 s^m + b_1 s^{m - 1} + \dots + b_m}{a_0 s^n + a_1 s^{n - 1} + \dots + a_n} = k^* \frac{\Pi_{i = 1}^m (s - z_i)}{\Pi_{j = 1}^n (s - p_j)}, \quad n > m $$

\begin{itemize}
\item 极点$ p_j $：特征根，决定模态。
\item 零点$ z_i $：不决定模态，影响响应比重、曲线形状，距原点近、极点远的影响大。
\item 传递系数/根轨迹增益$ k^* $。
\end{itemize}

\begin{itemize}
\item 混合节点：流向该节点信号代数和。
\item 支路：方向 + 传递函数，相当于乘法器，保证信号单向传递。
\item 前向通路：输入到输出，不重复过节点。
\item 回路：起止于同一节点，不重复过节点，其所有支路增益的乘积为回路增益。无公共节点的两个回路为不接触回路。
\end{itemize}

\item 源点、阱点不要和相邻节点合并。
\end{itemize}
\end{minipage}
%------------------------------------------------
\paragraph{\textcolor{red}{梅森（Mason）公式}}\tip{梅森公式}
$$ \textcolor{red}{P = \frac{1}{\underbrace{\Delta}_{\text{简单回路}}} \sum_{k = 1}^n \underbrace{P_k}_{\text{前向通路}} \Delta_k, \quad \Delta = 1 - \sum L_i + \sum L_i L_j - \sum L_i L_j L_r + \dots} $$

其中，$ \Delta $是特征式，$ P_k $是第$ k $条前向通路的传递函数，$ \Delta_k $是余子式，为$ \Delta $略去$ P_k $接触项。
\begin{itemize}
\item 基于克莱姆法则直接求传递函数。    
\item 绘制信号流图时，两节点间增益为$ 1 $，若合并后两回路接触，则不可合并。
\item 求解的是输出对输入的增益，若求解输出对中间变量的增益，可省去前面输入支路进行计算，也可使用输出、中间变量对输入的比值。
\end{itemize}

\begin{itemize}
\item \textcolor{red}{第一列元素}变号次数为右半平面根数，\textcolor{red}{全同号则稳定}。
\item \textcolor{red}{第一列有$ 0 $}，用$ \epsilon \to 0 $代替。
\item \textcolor{red}{某行全为$ 0 $}，用上一行作为辅助方程（系统不稳定，判断根的情况）求导：
$$ \begin{array}{ccccc} s^n & a & b & c & \cdots \\ s^{n - 1} & na & (n - 2)b & (n - 4)c & \cdots \end{array} $$
\item 求使系统稳定的$ k $：拓展为使根在$ -x $左侧（预留波动空间）时，用$ s = s' - x $代换，对$ s' $求解劳斯判据。
\end{itemize}

\section{根轨迹分析}
%------------------------------------------------
\subsection[概念]{概念}
系统参数连续变化时，特征方程的根在跟平面留下的轨迹，与系统性能相关。
%------------------------------------------------
\paragraph{推导}
$$ \Phi(s) = \frac{G(s)}{1 + G(s)H(s)}, \quad G(s)H(s) = k_G^* \frac{\Pi_{i = 1}^f (s - z_i)}{\Pi_{j = 1}^h (s - p_j)} k_H^* \frac{\Pi_{i = 1}^l (s - z_i)}{\Pi_{j = 1}^h (s - p_j)} = k^* \frac{\Pi_{i = 1}^m (s - z_i)}{\Pi_{j = 1}^n (s - p_j)} $$

其中，$ k_G^* $未前向通路根轨迹增益，$ k_H^* $为反馈通路根轨迹增益，$ k^* $为开环系统根轨迹增益，$ z_i, p_j $为开环零极点，有：
$$ \Phi(s) = 0 \Leftrightarrow G(s)H(s) = -1 = e^{j(2k + 1)\pi} $$

有稳定条件：
$$ \begin{cases} \text{\textcolor{red}{相角条件}（充要条件）：}\sum_{i = 1}^m \angle (s - z_i) - \sum_{j = 1}^n \angle (s - p_j) = (2k + 1)\pi \\ \text{模值条件：}k^* = \frac{\Pi_{j = 1}^n |(s - p_j)|}{\Pi_{i = 1}^m |(s - z_i)|} \end{cases} $$
%------------------------------------------------
\subsection[绘制]{绘制\tip{根轨迹绘制}}
\begin{enumerate}
\item 
\begin{itemize}
\item \textcolor{red}{起点}：$ n $个\textcolor{red}{极点}$ p_j $，$ (m - n) $个\textcolor{red}{无穷远处}。
\item \textcolor{red}{终点}：$ m $个\textcolor{red}{零点}$ z_i $，$ (n - m) $个\textcolor{red}{无穷远处}。
\end{itemize}
\item \textcolor{red}{分支数}：$ max(m, n) $，等于闭环极点数。
\begin{itemize}
\item 对称性：对称于实轴。
\item 连续性。
\end{itemize}
\item \textcolor{red}{渐近线}：$ n - m $条。
\begin{itemize}
\item 与实轴\textcolor{red}{夹角}：$ \textcolor{red}{\varphi_a = \frac{(2k + 1)\pi}{n - m} \in (-\pi, \pi)} $。
\item 与实轴\textcolor{red}{交点}：$ \textcolor{red}{\sigma_a = \frac{\sum_{j = 1}^n p_j - \sum_{i = 1}^m z_i}{n - m}} $。
\end{itemize}
\item \textcolor{red}{实轴部分}：左侧点角度为$ 0 $，右侧点角度为$ \pi $，一对共轭点角度为$ 2\pi $。
$$ \sum_{i = 1}^m \angle (s - z_i) - \sum_{j = 1}^n \angle (s - p_j) = (2k + 1)\pi $$ 

确保\textcolor{red}{右侧有奇数个零极点}的线段组合。
\item \textcolor{red}{分离点}与会合点：闭环极点为重根。
\begin{itemize}
\item \textcolor{red}{坐标$ d $}：由$ \textcolor{red}{\sum_{i = 1}^m \frac{1}{d - z_i} = \sum_{j = 1}^n \frac{1}{d - p_j}} $解得。
\item \textcolor{red}{角度}：$ \textcolor{red}{\frac{(2k + 1)\pi}{l}} $，$ l $为该处重根数。
\end{itemize}
\item 起始角与终止角：$ \varphi_{z x} $是$ z $到出射角的幅角，$ \theta_{p x} $是$ p $到入射角的幅角。
\begin{itemize}
\item \textcolor{red}{起始/出射角$ \theta_{pj} $}：离开极点切线与正实轴夹角。 
$$ \textcolor{red}{\theta_{pj} = (2k + 1)\pi + \sum_{a = 1}^m \varphi_{z_a p_j} - \sum_{b = 1, b \neq j}^n \theta_{p_b p_j}} $$
\item \textcolor{red}{终止/入射角$ \varphi_{zi} $}：抵达零点切线与正实轴夹角。
$$ \textcolor{red}{\varphi_{zi} = (2k + 1)\pi + \sum_{a = 1}^n \theta_{p_a z_i} - \sum_{b = 1, b \neq i}^m \varphi_{z_b z_i}} $$
\end{itemize}
\item \textcolor{red}{虚轴交点}：
$$ 
\textcolor{red}{1 + G(j \omega)H(j \omega) = 0} 
\Rightarrow \begin{cases} Im[1 + G(j \omega)H(j \omega)] = 0 \\ Re[1 + G(j \omega)H(j \omega)] = 0 \end{cases}
\overset{\text{求解}}{\Longrightarrow} \begin{cases} \textcolor{red}{\text{交点}k_{rc}} \\ \textcolor{red}{\text{临界增益}\omega_c} \end{cases}
$$

需注意$ k_{rc} $的多解舍解判断。
\item \textcolor{red}{根之和}：特征方程次高项系数为点集和，与虚轴交点结合计算其它交点。
$$ n - m \geq 2, \quad \textcolor{red}{\sum_{i = 1}^m z_i = \sum_{j = 1}^n p_j = -a_1} $$
\end{enumerate}
%------------------------------------------------
\subsection[广义根轨迹]{广义根轨迹}
%------------------------------------------------
\paragraph{\textcolor{red}{参数根轨迹}}
零极点与参数有关。
$$ 1 + G(s)H(s) = 0 \overset{\text{提}k}{\Longrightarrow} k A(s) + B(s) = 0 \Rightarrow k \frac{B(s)}{A(s)} + 1 = 0 $$

$ k \frac{B(s)}{A(s)} $是$ G(s)H(s) $的等效传递函数。等效单位反馈系统的闭环极点相同，闭环零点一般不同。
%------------------------------------------------
\paragraph{\textcolor{red}{零度根轨迹}}
相角遵循$ 2k\pi $条件的系统（非最小相位系统、正反馈系统）。

对上述步骤的公式进行如下替换：
$$ \begin{cases} 1 + G(s)H(s) = 0 \Rightarrow 1 - G(s)H(s) = 0 \\ (2k + 1)\pi \Rightarrow 2k \pi \\ \text{实轴部分：奇数个} \Rightarrow \text{偶数个} \end{cases} $$
%------------------------------------------------
\paragraph{其它}
\begin{itemize}
\item 主导极点：最接近虚轴，不十分接近闭环零点的闭环极点，可用其将高阶近似为低阶。
\item 偶极子：一对极近的闭环零极点。
\end{itemize}

\subsection[稳定性分析]{稳定性分析}
%------------------------------------------------
\subsubsection[幅角原理与奈式路径]{幅角原理与奈式路径\tip{幅角原理与奈式路径}}
%------------------------------------------------
\paragraph{\textcolor{red}{幅角原理}}
$ s $平面上闭合曲线$ \Gamma $包围$ F(s) $的$ P $个极点和$ Z $个零点，$ \Gamma $顺时针转一周，$ F(s) $平面上闭合曲线绕原点（$ \Gamma_{G_k} $绕$ (-1, j0) $）逆时针转$ R = P - Z $圈。
\begin{align*}
&G_k(s) = G(s)H(s) = \frac{N(s)}{D(s)}, \quad G_B(s) = \frac{G_k(s)}{1 + G_k(s)} \\
&F(s) = 1 + G_k(s) = \frac{D(s) + N(s)}{D(s)} 
\end{align*}

其中$ F(s) $的零点是$ G_B(s) $的极点（$ Z $个），极点是$ G_k(s) $的极点（$ P $个）。

\textcolor{red}{稳定$ \Leftrightarrow $$ F(s) $在右半平面$ Z = P - R = 0 $。}
%------------------------------------------------
\paragraph{\textcolor{red}{奈式路径}$ \Gamma_{G_k} $}
$ \omega \in [0, \infty) $
\begin{itemize}
\item \textcolor{red}{虚轴}：$ s = j \omega, \quad \omega \in [0, \infty) $，去掉极点对应的部分，为幅相曲线。
\item \textcolor{red}{无穷}：$ s = \infty e^{j \theta}, \quad \theta \in [90^\circ, 0] $，是一个点。
\item \textcolor{red}{极点}：
\begin{itemize}
\item 原点：$ s = \epsilon e^{j \theta}, \quad \epsilon = 0^+, \quad \theta \in [0, 90^\circ] \Rightarrow G(s)H(s) = \infty e^{-j[\underbrace{v}_{\text{积分次数}} \theta + \angle G_0(0)]} $，起于$ \omega = 0^+ $，$ R = \infty $，圆心角为$ v \times 90^\circ $的顺时针圆弧，交幅相曲线于$ G(j 0^+)H(j 0^+) $。
\item 虚轴对称点：$ s = j \omega_n + \epsilon e^{j \theta} \Rightarrow G(s)H(s) = \infty G_0(j \omega_n) e^{-j(\theta + 90^\circ)v} $，起于$ G(j \omega_n^-)H(j \omega_n^-) $止于$ G(j \omega_n^+)H(j \omega_n^+) $，$ R = \infty $，圆心角为$ v \times 180^\circ $的顺时针圆弧。
\end{itemize}
\end{itemize}
%------------------------------------------------
\paragraph{\textcolor{red}{穿越次数}}
$ \Gamma_{G_k} $包围$ \textcolor{red}{(-1, j 0)} $的圈数$ \textcolor{red}{R = 2N = 2(N^+ - N^-)} $。
\begin{itemize}
\item 正穿越$ N^+ $为$ (-1, j 0) $左侧由上至下\textcolor{red}{逆时针}穿过，负穿越$ N^- $反之。
\item 半次：起/止于$ (-1, j 0) $左侧。
\item $ (-1, j 0) $处不计数，存在纯虚根，可能临界稳定。
\end{itemize}
%------------------------------------------------
\paragraph{\textcolor{red}{奈奎斯特稳定判据}}
\textcolor{red}{稳定$ \Leftrightarrow $$ \Gamma_{G_k} $不穿过$ (-1, j 0) $，且逆时针包围$ (-1, j 0) $的圈数$ R $等于开环传递函数的正实部极点数$ P $}。
%------------------------------------------------
\paragraph{对数频率稳定判据}
系统稳定$ \Leftrightarrow $$ \varphi(\omega_c) \neq (2k + 1)\pi, \quad k = 1, 2, \dots, $，$ L(\omega) > 0 $时，曲线穿越$ (2k + 1)\pi $线次数$ Z = P - 2R = 0 $。
%------------------------------------------------
\paragraph{类型}
\begin{itemize}
\item 条件稳定：系数影响稳定性。
\item 结构不稳定：不稳定性与系数无关。
\end{itemize}

\item 前馈：校正在主反馈外，分为滤波和补偿。
\item 复合：按扰动/输出进行补偿。
\end{itemize}

\paragraph{\textcolor{red}{方法}}
\begin{itemize}
\item \textcolor{red}{级数求和法（定义法）}：
$$ E(z) = \sum_{n = 0}^{\infty} e(nT)z^{-n} = e(0) + e(T)z^{-1} + \dots + e(nT)z^{-n} + \dots $$
\item \textcolor{red}{部分分式法}：$ e(t) \Rightarrow E(s) \Rightarrow $部分分式和$ \Rightarrow E(z) $。
\end{itemize}
%------------------------------------------------
\paragraph{\textcolor{red}{性质}}
\begin{itemize}
\item 线性：$ Z(\alpha_1 e_1 \pm \alpha_2 e_2) = \alpha_1 Z(e_1) \pm \alpha_2 Z(e_2) = \alpha_1 E_1(z) \pm \alpha_2 E_2(z) $。
\item \textcolor{red}{终值定理}：$ e(\infty) = \lim_{z \to \textcolor{red}{1}} \textcolor{red}{(z - 1)}E(z) $。
\item \textcolor{red}{位移定理}：
\begin{itemize}
\item 实数：$ Z[e^*(t - kT)] = z^{-k}E(z), \quad Z[e^*(t + kT)] = z^{k}[E(z) - \sum_{n = 0}^{k - 1} e(nT)z^{-n}] $
\item 复数：$ Z[e^*(t)e^{\mp \alpha T}] = E(z e^{\pm \alpha T}) $。
\end{itemize}
\item \textcolor{red}{卷积定理}：$ x(nT) \cdot y(nT) = \sum_{k = 0}^{\infty} x(kT)y[(n - kT)], G(z) = X(z) \cdot Y(z) $。

\subsection[线性定常系统]{线性定常系统（LTI）\tip{线性定常系统}}
$$ \text{齐次} \begin{cases} \dot{x} = A x(t) \\ x(t_0) = x_0 \end{cases} \quad \text{非齐次} \begin{cases} \dot{x} = A x(t) + B u(t) \\ x(t_0) = x_0 \end{cases} $$
%------------------------------------------------
\paragraph{状态变量$ \textcolor{red}{x(t)} $}
\begin{itemize}
\item 幂级数（齐次）：
\begin{align*}
&\begin{cases} x(t) = b_0 + b_1 t + \dots + b_n t^n + \dots \\ \dot{x}(t) = b_1 + 2b_2 t + \dots + n b_n t^{n - 1} + \dots = A x(t) \end{cases} \\
\Rightarrow &b_1 = A b_0, \quad b_2 = \frac{1}{2} A b_1 = \frac{1}{2!} A^2 b_0, \dots, \quad b_n = \frac{1}{n} A b_{n - 1}  = \frac{1}{n!} A^n b_0 \\
\Rightarrow &x(t) = b_0 \sum_{n = 0}^{\infty} \frac{(At)^n}{n!} = b_0 e^{At} = x_0 e^{At}
\end{align*}
\item 积分（非齐次）：
\begin{align*}
&e^{-At}[\dot{x}(t) - A x(t)] = e^{-At} B u(t) \\
\overset{\text{定积分还原}}{\Longrightarrow} &[e^{-At}x(t)]|_{t_0}^{t} = \int_{t_0}^{t} e^{-A\tau} B u(\tau) d \tau = e^{-At} x(t) - e^{-A t_0} x(t_0) \\
\Rightarrow &x(t) = \underbrace{x(t_0) e^{A(t - t_0)}}_{\text{初始状态引起的自由运动}} + \underbrace{\int_{t_0}^{t} e^{A(t - \tau)} B u(\tau) d \tau}_{\text{控制激励引起的强制运动}}
\end{align*} 
\item 拉氏变换（非齐次）：
\begin{align*}
&L[\dot{x}(t)] = L[A x(t) + B u(t)] \overset{\text{拉氏变换}}{\Longrightarrow} s X(s) - x(0) = A X(s) + B U(s) \\
\Rightarrow &(sI - A) X(s) = x(0) + B U(s) \overset{\text{拉氏反变换}}{\Longrightarrow} x(t) = L^{-1} [(sI - A)^{-1}] x(0) + \underbrace{L^{-1} [(sI - A)^{-1} B U(s)]}_{\text{卷积}} \\
\Rightarrow &\textcolor{red}{x(t) = e^{At} x(0) + \int_{0}^{t} e^{A(t)} B u(t - \tau) d \tau}
\end{align*}    
\end{itemize}
%------------------------------------------------
\paragraph{状态转移矩阵（矩阵指数函数）$ \textcolor{red}{\Phi(t) = e^{A(t)}} $}
\begin{itemize}
\item 幂级数：$ e^{A(t)} = I + At + \frac{1}{2!}(At)^2 + \dots + \frac{1}{n!}(At)^n + \dots = \sum_{n = 0}^{\infty} \frac{(At)^n}{n!} $。
\item \textcolor{red}{拉氏变换}：$ \textcolor{red}{e^{A(t)} = L^{-1} [(sI - A)^{-1}]} $。
\item \textcolor{red}{凯莱哈密顿定理}：
\begin{itemize}
\item 原理
\begin{itemize}
\item 矩阵的幂可以由低阶幂线性表示（递推证明）：$ A^k = \sum_{m = 0}^{n - 1} \alpha_m A^m, \quad k \geq m $。
\item $ e^{A(t)} $展开：$ e^{A(t)} = \sum_{m = 0}^{n - 1} \alpha_m(t) A^m $。
\item 特征多项式：$ f(\lambda) = |\lambda I - A| = \lambda^n + a_{n - 1} \lambda^{n - 1} + \dots + a_1 \lambda + a_0 \Rightarrow f(A) = 0 $。
\end{itemize}
\item \textcolor{red}{用法1}：
\begin{enumerate}
\item 求特征根$ \lambda_i $：解$ f(\lambda) = 0 $。
\item 由$ n $个方程$ \textcolor{red}{e^{\lambda_i(t)} = \sum_{m = 0}^{n - 1} \alpha_m(t) \lambda_i^m} $，解得$ n $个$ \alpha(t) $。
\item 代回$ e^{A(t)} = \sum_{m = 0}^{n - 1} \alpha_m(t) A^m $。
\end{enumerate}
\item 用法2：
$$ f(A) = \sum_{i = 0}^{n} a_i A^i = 0 \Rightarrow a_n A^n = -\sum_{i = 0}^{n - 1} a_i A^i \underset{{\text{代入}A^n}}{\overset{\text{两端乘}A}{\Longrightarrow}} A^{n + 1} = \sum_{i = 0}^{n - 1} a_i' A^i $$
\end{itemize}
\end{itemize}

\item 不稳定：$ \exists \epsilon > 0, \quad \forall \delta > 0 $，针对$ s(\delta) $，$ \exists x_0 $使其发出的轨迹超出$ s(\epsilon) $。
\end{itemize}

\item 还原：$ \textcolor{red}{k' = k T^{-1}} $。
\end{enumerate}
\item \textcolor{red}{法2}
\begin{enumerate}

\item 计算极点：$ \textcolor{red}{|sI - (A + B k)| = s^n + a_{n - 1} s^{n - 1} + \dots + a_1 s + a_0} $。

\item 两式\textcolor{red}{对应位置系数相等}，得到$ k $。
\end{enumerate}
\item 法3：阿克曼公式，高阶单输入，计算机使用。
\end{itemize}

\paragraph{\textcolor{red}{解李雅普诺夫方程}}
$ (A, B) $能控、$ (\overline{k}, F) $能观测$ \Leftrightarrow $$ A P - P F = B \overline{k} $有唯一非奇异解（SI为充要，MI为必要）。

\item 选取$ F_{n \times n} $（如能观测标准型伴随阵），包含目标极点但不包含$ A $的特征根。
\item 任取$ \overline{k}_{m \times n} $（如$ \begin{bmatrix} 0 & \cdots & 0 & 1 \end{bmatrix} $），使$ (\overline{k}, F) $能观测。
\item 解$ A P - P F = B \overline{k} $的唯一解$ P $。若$ P $奇异则选取其它$ \overline{k} $重新计算，否则$ k = \overline{k} P $。
\end{enumerate}

\subsection[解耦]{\textcolor{red}{解耦}\tip{解耦}}
使闭环系统中每个输入仅影响一个输出，MIMO$ \Rightarrow $多SISO。
$$ \begin{cases} \dot{x} = A x + B u \\ y = C x \end{cases} \overset{u = k_1 x + k_2 v}{\Longrightarrow} \overline{G}(s) = \underbrace{C (sI - A + B k_1)^{-1} B}_{\textcolor{red}{\text{非奇异对角阵}}} k_2 $$
%------------------------------------------------
\paragraph{状态反馈解耦}
\begin{enumerate}
\item 求$ C $第$ i $行对应的$ \sigma_i $，其为使$ C_i A^{j - 1} B $非奇异的最小正整数，不超过$ n - 1 $。
$$ \textcolor{red}{\sigma_i = \begin{cases} \min\{j|C_i A^{j - 1} B \neq 0\} \\ n - 1, \quad \forall j < n, \quad C_i A^{j - 1} B = 0 \end{cases}} $$
\item 若$ \textcolor{red}{E = \begin{bmatrix} C_1 A^{\sigma_1 - 1} B \\ \cdots \\ C_m A^{\sigma_m - 1} B \end{bmatrix}} $非奇异，则$ u = k_1 x + k_2 v $可解耦。
\item $ L = \begin{bmatrix} C_1 A^{\sigma_1} \\ \cdots \\ C_m A^{\sigma_m} \end{bmatrix}, \quad k_1 = -E^{-1} L, \quad k_2 = E^{-1} $。
\item 解耦后，$ \textcolor{red}{C (sI - A + B k_1)^{-1} B = diag(\frac{1}{s^{\sigma_1}}, \dots, \frac{1}{s^{\sigma_m}})} $。
\end{enumerate}

\subsubsection[降维观测器]{\textcolor{red}{降维观测器}}
部分状态变量（$ r $维，对应$ x_2 $）可用作反馈，不需要估计，只需$ n - r $阶观测器。
\begin{enumerate}
\item 改写系统，使输出由部分状态变量表示。
\begin{enumerate}
\item 任取$ D $，使$ Q = \begin{bmatrix} D \\ C \end{bmatrix} $可逆，作变换$ x = Q^{-1} x' $，得：
$$ \begin{cases} \dot{x} = \begin{bmatrix} \dot{x}_1 \\ \dot{x}_2 \end{bmatrix} = \begin{bmatrix} A_{11} & A_{12} \\ A_{21} & A_{22} \end{bmatrix} \begin{bmatrix} x_1 \\ x_2 \end{bmatrix} + \begin{bmatrix} B_1 \\ B_2 \end{bmatrix} u \\ y = \begin{bmatrix} 0 & I_r \end{bmatrix} \begin{bmatrix} x_1 \\ x_2 \end{bmatrix} \Rightarrow y = x_2 \end{cases} $$
\item 若以上系统完全能观测，则以下子系统也完全能观测：
$$ \begin{cases} \dot{x}_1 = A_{11} x_1 + A_{12} y + B_1 u = A_{11} x_1 + v \\ \overline{y} = \dot{y} - A_{22} y - B_2 u = A_{21} x_1 \end{cases} $$
\end{enumerate}
\item 为$ \dot{z} = (A_{11} + G A_{21}) z + \underbrace{A_{12} y + B_1 u}_{v} - G(\underbrace{\dot{y} - A_{22} y - B_2 u}_{\overline{y}}) $设计观测器。
\begin{enumerate}
\item 即\textcolor{red}{为$ \dot{w} = \dot{z} + G \dot{y} = \underbrace{(A_{11} + G A_{21})}_{\text{满足期望极点}} w + (B_1 + G B_2)u + [A_{12} + G A_{22} - (A_{11} + G A_{21})G]y $配置极点}。
\item $ x $的估计为$ \textcolor{red}{\hat{x} = \begin{bmatrix} z \\ y \end{bmatrix} = \begin{bmatrix} I \\ 0 \end{bmatrix} w + \begin{bmatrix} -G \\ I \end{bmatrix} y} $。
\item \textcolor{red}{返回原系统}。
\end{enumerate}
\end{enumerate}

\subsection[带观测器的状态反馈器]{\textcolor{red}{带观测器的状态反馈器}\tip{带观测器的状态反馈器（分离定理）}}
$$ \begin{cases} \dot{x} = A x + B u \\ y = C x \end{cases} \text{与} \begin{cases} \dot{\hat{x}} = (A + G C)\hat{x} + B u - G y \\ u = k \hat{x} + v \end{cases} $$

组合得到：
$$ \begin{cases} \begin{bmatrix} \dot{x} \\ \dot{\hat{x}} \end{bmatrix} = \underbrace{\begin{bmatrix} A & B k \\ -G C & A + G C + B k \end{bmatrix}}_{A'} \begin{bmatrix} x \\ \hat{x} \end{bmatrix} + \underbrace{\begin{bmatrix} B \\ B \end{bmatrix}}_{B'} v \\ y = \underbrace{\begin{bmatrix} C & 0 \end{bmatrix}}_{C'} \begin{bmatrix} x \\ \hat{x} \end{bmatrix} \end{cases} $$

经过线性变换$ \begin{bmatrix} I_n & 0 \\ I_n & -I_n \end{bmatrix} $，有$ det(sI - A') = det[sI - (A + B k)] \cdot det[sI - (A + G C)] $。
\begin{itemize}
\item \textcolor{red}{分离定理}：\textcolor{red}{极点上，带观测器的状态反馈器 = 原闭环系统 + 观测器}。
\item $ \textcolor{red}{G(s)} $：$ \textcolor{red}{C'(sI - A')^{-1}B' = C[sI - (A + B k)]^{-1}B} $。
\item 极点位置确定：配置在闭环极点远端，使估计尽快逼近实际，兼顾状态估计误差的衰减速率（快于系统响应速率）和观测器的抗干扰能力。
\end{itemize}

\subsubsection[泛函]{\textcolor{red}{泛函}\tip{泛函}}
%------------------------------------------------
\paragraph{函数与泛函}
\begin{itemize}
\item $ y = y(x) $，$ y $是$ x $的函数；$ J = J[y(x)] $，$ J $为依赖$ y(x) $的泛函。
\item 容许函数类：满足条件的函数，差值为宗量变分$ \delta y(x) = y(x) - y_0(x) $。
\item 线性：$ J[c_1 y_1(x) + c_2 y_2(x)] = c_1 J[y_1(x)] + c_2 J[y_2(x)] $。
\end{itemize}
%------------------------------------------------
\paragraph{\textcolor{red}{求解积分型指标泛函}$ J = \int_{t_0}^{t_f} L[x(t), \dot{x}(t), t] d t $}
以$ J = \int_{x_0}^{x_1} y^2(x) d x $为例。
\begin{itemize}
\begin{minipage}{0.5\textwidth}
\item 法1：变分取线性主部$ \delta J $
\begin{align*} \Delta J &= \int_{x_0}^{x_1} [y(x) + \delta y(x)]^2 d x - \int_{x_0}^{x_1} y^2(x) d x \\ &= \int_{x_0}^{x_1} \{\underbrace{[2y(x) \delta y(x)]}_{\text{线性主部}\delta J} + [\delta y(x)]^2\} d x \end{align*}
\end{minipage}
\begin{minipage}{0.5\textwidth}
\item 法2：微分
\begin{align*} 
\Delta J &= \frac{\partial}{\partial \epsilon} J[y(x) + \epsilon \delta y(x)]|_{\epsilon = 0} \\
&= \int_{x_0}^{x_1} \frac{\partial}{\partial \epsilon}[y(x) + \epsilon \delta y(x)]^2 d x|_{\epsilon = 0} \\
&= \int_{x_0}^{x_1} 2[y(x) + \epsilon \delta y(x)] \delta y(x) d x|_{\epsilon = 0}
\end{align*}
\end{minipage}
\end{itemize}

\subsubsection[欧拉拉格朗日方程]{\textcolor{red}{欧拉拉格朗日方程$ \frac{\partial f}{\partial x} - \frac{d}{d t}(\frac{\partial f}{\partial \dot{x}}) = 0 $}}

\paragraph{变分预备定理}
$ y(t) $在$ [t_0, t_f] $上连续，$ y(t_0) = y(t_f) = 0 $，$ g(t) $在$ [t_0, t_f] $上连续，有$ \int_{t_0}^{t_f} g(t) y(t) d t = 0 $，则$ [t_0, t_f] $上必有$ g(t) \equiv 0 $。
%------------------------------------------------
\paragraph{\textcolor{red}{简化解法}（必要条件）}\tip{欧拉拉格朗日方程}
$ u = g(x, \dot{x}), \quad J(u) = \int_{t_0}^{t_f} f(x, u) d t, \quad x(t_0) = x_0, \quad argmin_{u(t)} J(u) $。
\begin{enumerate}
\item 把$ u $代入$ f(x, u) $，求得$ f(x, \dot{x}) $。
\item $ \frac{\partial f}{\partial x} - \frac{d}{d t}(\frac{\partial f}{\partial \dot{x}}) = 0 $（\textcolor{red}{只求偏导}）。
\item \textcolor{red}{正则方程}：
\begin{itemize}
\item 边界条件：$ x(t_0) = x_0 $。
\item 横截条件：
\begin{itemize}
\item $ x(t_f) = x_f $。
\item $ t_f $已知，$ x(t_f) $自由：$ \frac{\partial f}{\partial \dot{x}}|_{t_f} = 0 $。
\item $ t_f, x(t_f) $自由：$ \frac{\partial f}{\partial \dot{x}}|_{t_f} = 0, \quad f(x, \dot{x}, t)|_{t_f} = 0 $。
\item $ x(t_f) $落在$ \varphi(t_f) $上：$ \{f(x, \dot{x}, t) + [\dot{\varphi}(t) - \dot{x}]\frac{\partial f}{\partial \dot{x}}\}|_{t_f} = 0 $（联立$ \varphi(t_f) $求$ t_f $）。
\end{itemize}
\end{itemize}
\item 解通解$ x(t) $，代入正则方程确定。
\item 代回$ u = g(x, \dot{x}) $求$ u(t) $。
\end{enumerate}
%------------------------------------------------
\paragraph{一般泛函$ J(x) = \theta[x(t_f), t_f] + \int_{t_0}^{t_f} f(x, \dot{x}, t) d t $推导}
\begin{align*}
&\Phi(\epsilon) = \theta[x(t_f + \epsilon \delta t_f) + \epsilon \delta x(t_f + \epsilon \delta t_f), t_f + \epsilon \delta t_f] + \int_{t_0}^{t_f + \epsilon \delta t_f} f(x + \epsilon \delta x, \dot{x} + \epsilon \delta \dot{x}, t) dt \\
&\frac{d \Phi}{d t}|_{\epsilon = 0} = \frac{\partial \theta[x(t_f), t_f]}{\partial x(t_f)}[\dot{x}(t_f) \delta t_f + \delta x(t_f)] + \frac{\partial \theta[x(t_f), t_f]}{\partial t_f} \delta t_f \\
&+ \int_{t_0}^{t_f}(\frac{\partial f}{\partial x} - \frac{d}{d t}\frac{\partial f}{\partial \dot{x}}) \delta x dt + (\frac{\partial f}{\partial \dot{x}} \delta x)|_{t_0}^{t_f} + f(x, \dot{x}, t)|_{t_f} \delta t_f \\
&\begin{cases} \delta x(t_f): \frac{\partial f}{\partial \dot{x}}|_{t_f} = -\frac{\partial \theta[x(t_f), t_f]}{\partial x(t_f)} \\ \delta t_f: \{f(x, \dot{x}, t) - \dot{x}\frac{\partial f}{\partial \dot{x}} + \frac{\partial \theta[x(t_f), t_f]}{\partial t_f}\}|_{t_f} = 0 \end{cases} \Rightarrow \text{变更}
\end{align*}

\subsubsection[哈密顿函数]{\textcolor{red}{哈密顿函数}（拉格朗日乘子法）}
%------------------------------------------------
\paragraph{推导}
$$ \dot{x} = f(x, u, t), \quad x(t_0) = x_0, \quad J(u) = \theta[x(t_f), t_f] + \int_{t_0}^{t_f} L(x, u, t) d t $$

引入拉格朗日乘子求极值：
$$ \overline{J}(x, u, \lambda) = \theta[x(t_f), t_f] + \int_{t_0}^{t_f}\{\textcolor{red}{\underbrace{L(x, u, t) + \lambda^T[f(x, u, t) - \dot{x}]}_{\text{哈密顿函数}H}}\}d t $$

对各变量求拉格朗日方程：
$$ \begin{cases} \frac{\partial H}{\partial x} - \frac{d}{d t}(\frac{\partial H}{\partial \dot{x}}) = 0 \\ \frac{\partial H}{\partial u} - \frac{d}{d t}(\frac{\partial H}{\partial \dot{u}}) = 0 \\ \frac{\partial H}{\partial \lambda} - \frac{d}{d t}(\frac{\partial H}{\partial \dot{\lambda}}) = 0 \end{cases} \Rightarrow \begin{cases} \textcolor{red}{\dot{\lambda} = -\frac{\partial H}{\partial x} \quad \text{协状态方程}} \\ \textcolor{red}{\frac{\partial H}{\partial u} = 0} \\ \textcolor{red}{\dot{x} = f(x, u, t) = \frac{\partial H}{\partial \lambda}} \end{cases} $$
%------------------------------------------------
\paragraph{\textcolor{red}{解法}}\tip{哈密顿函数（连续）}
\begin{enumerate}
\item 构造$ H(x, \lambda, u, t) = L(x, u, t) + \lambda^T f(x, u, t) $。
\item $ \frac{\partial H}{\partial u} = 0 $，得$ u^* = u(x, \lambda) $，代入$ f(x, u, t) $得到$ f(x, \lambda, t) $。
\item $ \begin{cases} \dot{\lambda} = \textcolor{red}{-}\frac{\partial H}{\partial x} \\ \dot{x} = f(x, \lambda, t) = \frac{\partial H}{\partial \lambda} \end{cases} $，\textcolor{red}{对向量的各元素分开列}。
\item \textcolor{red}{正则方程}：
\begin{itemize}
\item 边界条件：$ x(t_0) = x_0 $。
\item 横截条件：
\begin{itemize}
\item $ x(t_f) = x_f $。
\item $ t_f $已知，$ x(t_f) $自由：$ \textcolor{red}{\lambda(t_f) = \frac{\partial \theta[x(t_f), t_f]}{\partial x(t_f)}} $。
\item $ t_f, x(t_f) $自由：$ \lambda(t_f) = \frac{\partial \theta[x(t_f), t_f]}{\partial \textcolor{red}{x(t_f)}}, \quad H(x, \lambda, u, t)|_{t_f} + \frac{\partial \theta[x(t_f), t_f]}{\partial \textcolor{red}{t_f}} = 0 $。
\item $ N[x(t_f), t_f] = 0 $：$ \lambda(t_f) = \frac{\partial \theta[x(t_f), t_f]}{\partial x(t_f)} + \frac{\partial N^T[x(t_f), t_f]}{\partial x(t_f)} \cdot v, \quad H(x, \lambda, u, t)|_{t_f} + \frac{\partial \theta[x(t_f), t_f]}{\partial t_f} + \frac{\partial N^T[x(t_f), t_f]}{\partial t_f} \cdot v = 0 $，其中$ v $为$ r $维实向量，通过作比、代回$ N $中等方法消去。
\item $ H $不显式含$ t $：$ \dot{H} = \underbrace{(\frac{\partial H}{\partial x})^T}_{-\dot{\lambda}^T} \dot{x} + \underbrace{(\frac{\partial H}{\partial u})^T}_{0} \dot{u} + \underbrace{(\frac{\partial H}{\partial \lambda})^T}_{\dot{x}^T} \dot{\lambda} = 0, \quad H[x^*(t), u^*(t), \lambda^*(t)] $为常数。
\item $ t_f $自由，$ H $不依赖于$ t $，$ \theta, N $不依赖于$ H $：$ H[x(t_f), u(t_f), \lambda(t_f)] = 0 $沿最优轨线。
\end{itemize}
\end{itemize}
\end{enumerate}

\subsection[最大值原理]{\textcolor{red}{最大值原理}（\textcolor{red}{有约束}）}
$ u $受限，$ L $关于$ u $不可微，不存在偏导数。
%------------------------------------------------
\paragraph{解法}\tip{最大值原理（连续）}
\begin{enumerate}
\item 构造$ H(x, \lambda, u, t) = L(x, u, t) + \lambda^T f(x, u, t) $，求$ argmin_{u} H $，其为\textcolor{red}{$ u $关于$ \lambda $的函数}。
\item $ \begin{cases} \dot{\lambda} = \textcolor{red}{-}\frac{\partial H}{\partial x} \\ \dot{x} = f(x, \lambda, t) = \frac{\partial H}{\partial \lambda} \end{cases} $，解$ \lambda $关于$ t $的函数，得到\textcolor{red}{$ u $关于$ t $的函数}。
\item $ x^*(t) = e^{At} x_0 + \int_{t_0}^{t} e^{A(t - \tau)} B u(\tau) d \textcolor{red}{\tau} $，注意$ u $取值引起的讨论。
\item \textcolor{red}{正则方程}：同哈密顿函数。
\item 代回$ J(u) $，分段积分。
\end{enumerate}
%------------------------------------------------
\paragraph{最小时间控制（最速控制）}
\begin{itemize}
\item 条件：$ \dot{x} = A x(t) + B u(t), \quad A \in R^n, \quad B \in R^m $，$ A, B $为常阵，$ (A, B) $能控。$ x(t_0) = x_0, \quad x(t_f) = 0 $，$ t_f $自由，$ J = \int_{0}^{t_f} d t, |u_j(t)| \leq 1 $。
\item 推导：$ H(x, u, \lambda) = 1 + \lambda^T(A x + B u), \quad B = \begin{bmatrix} b_1 & b_2 & \cdots & b_m \end{bmatrix} $，设$ g_j(t) = \lambda^T(t) b_j $，有$ u_j^*(t) = -sgn\{g_j(t)\} $。
\item 
\begin{itemize}
\item \textcolor{red}{正常}：$ g_j(t) \neq 0 $，可用最大值原理确定$ u_j^*(t) $，\textcolor{red}{系统能控是系统正常的前提条件}。
\item \textcolor{red}{奇异}：$ g_j(t) = 0 $，只能取满足$ u $限制条件的任意值。
\item 判断：$ G_j = \begin{bmatrix} b_j & A b_j & \cdots & A^{n - 1} b_j \end{bmatrix}, \quad j = 1, 2, \dots, m $，$ G_j $非奇异则正常。
\end{itemize}
\item \textcolor{red}{BangBang控制原理}：定常线性系统正常，则可用最大值原理求解最速控制，在$ U $内从一个边界值切换到另一个。
\end{itemize}

\subsubsection[有限时间状态调节器]{\textcolor{red}{有限时间状态调节器}（\textcolor{red}{微分}）}
%------------------------------------------------
\paragraph{条件}
$ \dot{x} = A(t) x + B(t) u, \quad x(t_0) = x_0 $，\textcolor{red}{不要求$ (A, B) $能控}，$ F, Q $对称半正定，$ R $对称正定。
%------------------------------------------------
\paragraph{里卡蒂微分方程}~\\

列哈密顿函数：
\begin{align*} H &= \frac{1}{2}[x^T Q(t) x + u^T R(t) u] + \lambda^T[A(t) x + B(t) u] \\ \frac{\partial H}{\partial u} &= 0 \Rightarrow u^*(t) = -R^{-1}(t) B^T(t) \lambda(t) \end{align*}

得到：
\begin{align*} &\begin{cases} \dot{x} = A(t) x + B(t) u^*(t) \\ \dot{\lambda} = -\frac{\partial H}{\partial x} = -Q(t) x - A^T(t) \lambda \end{cases} \\ &\lambda(t_f) = \frac{\partial[\frac{1}{2} x^T(t_f) F x(t_f)]}{\partial x(t_f)} = F x(t_f) \\ &x(t_0) = x_0 \end{align*}

设$ \Phi(t, t_0) = \begin{bmatrix} \phi_{11}(t, t_0) & \phi_{12}(t, t_0) \\ \phi_{21}(t, t_0) & \phi_{22}(t, t_0) \end{bmatrix} $，有：
$$ \lambda(t) = \underbrace{[\phi_{22}(t_f, t) - F \phi_{12}(t_f, t)]^{-1} [F \phi_{11}(t_f, t) - \phi_{21}(t_f, t)]}_{k(t) \text{反馈增益矩阵} P(t) \text{，对称半正定}} x(t) $$

有$ \textcolor{red}{u(t) = -R^{-1}(t) B^T(t) P(t) x(t)} $。

对$ P(t) $求关于$ t $的导数，即为里卡蒂微分方程：
$$ \textcolor{red}{\frac{d P}{d t} = -P(t) A(t) - A^T(t) P(t) + P(t) B(t) R^{-1}(t) B^T(t) P(t) - Q(t)} $$

根据$ \lambda(t) $的函数式，$ \lambda(t_f) = P(t_f) x(t_f) = F x(t_f) $，有$ \textcolor{red}{P(t_f) = F} $。
%------------------------------------------------
\paragraph{\textcolor{red}{解法}}\tip{里卡蒂微分方程（有限时间状态调节器）}
\begin{enumerate}
\item 提取$ A, B, Q, R, F $阵，设对称的未知阵$ P(t) $。
\item 根据边界条件$ P(t_f, t_f) = F $解里卡蒂微分方程，得到的\textcolor{red}{$ P $应是半正定的}。
$$ \frac{d P}{d t} = -P(t) A(t) - A^T(t) P(t) + P(t) B(t) R^{-1}(t) B^T(t) P(t) - Q(t) $$
\item $ u^*(t) = -R^{-1}(t) B^T(t) P(t) x(t) $。
\end{enumerate}
%------------------------------------------------
\paragraph{定理}
\begin{itemize}
\item 方程与解：$ u^*(t) = -R^{-1}(t) B^T(t) P(t, t_f) x(t) $，其中$ P(t, t_f) $是以下方程组的解，$ P, Q $的（半）正定性一致。
$$ \begin{cases} \frac{d P}{d t} = -P(t) A(t) - A^T(t) P(t) + P(t) B(t) R^{-1}(t) B^T(t) P(t) - Q(t) \\ P(t_f, t_f) = F \end{cases} $$
\item \textcolor{red}{最优解}：$ J^*[x(t_0), t_0]_{min} = \frac{1}{2} x^T(t_0) P(t_0, t_f) x(t_0) $。
\item 最优轨线$ \begin{cases} \frac{d x}{d t} = [A(t) - B(t) R^{-1}(t) B^T(t) P(t_f, t_f)] x(t) \\ x(t_0) = x_0 \end{cases} $。
\end{itemize}

\subsubsection[非时变状态调节器]{\textcolor{red}{非时变状态调节器}（\textcolor{red}{常阵}）}
\begin{itemize}
\item 条件：$ \dot{x} = A x + B u, \quad x(t_0) = x_0, \quad \textcolor{red}{t_f = \infty} $，\textcolor{red}{$ (A, B) $完全能控}，$ Q $对称半正定，$ R $对称正定。
\item 常阵$ \hat{P} $：$ \hat{P} = \lim_{t \to \infty} P(t, t_f) = \lim_{t \to -\infty} P(t_0, t_f) $，$ \hat{p} $正定$ \Leftrightarrow $$ (C, A) $能观测。
\item 代数里卡蒂方程：$ \textcolor{red}{-\hat{P} A - A^T \hat{P} + \hat{P} B R^{-1} B^T \hat{P} - Q = 0} $。
\end{itemize}
%------------------------------------------------
\paragraph{解法}
\begin{itemize}
\item \textcolor{red}{法1}\tip{代数里卡蒂方程（非时变状态调节器）}：求代数里卡蒂方程。
\begin{enumerate}
\item 提取$ A, B, Q, R, F $阵，设对称的未知阵$ P $。
\item 解代数里卡蒂方程，得到的$ P $应是正定的。
$$ -\hat{P} A - A^T \hat{P} + \hat{P} B R^{-1} B^T \hat{P} - Q = 0 $$
\item $ u = -R^{-1} B^T P x $。
\end{enumerate}
\item 法2：求里卡蒂微分方程稳态解。
$$ \begin{cases} \frac{d P}{d t} = -P A - A^T P + P B R^{-1} B^T P - Q \\ P(t_0) = 0 \end{cases} $$
\item 法3：哈密顿矩阵法。
\begin{itemize}
\item 原理：

$ u^*(t) = - R^{-1} B^T \hat{P} x(t) \Rightarrow \begin{cases} \dot{x} = \underbrace{(A - B R^{-1} B^T \hat{P})}_{\overline{A}} x \\ x(t_0) = x_0 \end{cases} $，解得$ \overline{A} $的特征值$ \lambda $和特征根$ \eta $，有$ \lambda \eta = \overline{A} \eta $。构造哈密顿矩阵$ H_{2n \times 2n} $，特征值为$ \pm \lambda $（$ 2n $项）。

令$ \xi = \hat{P} \eta, \quad \lambda \xi = (\hat{P} A - \hat{P} B R^{-1} B^T \hat{P}) \eta = (-Q - A^T \hat{P}) \eta = -Q \eta - A^T \xi $，有：
$$ \lambda \begin{bmatrix} \eta \\ \xi \end{bmatrix} = \underbrace{\begin{bmatrix} A & -B R^{-1} B^T \\ -Q & -A^T \end{bmatrix}}_{H} \begin{bmatrix} \eta \\ \xi \end{bmatrix} $$

取$ T = \begin{bmatrix} I_n & 0 \\ \hat{P} & I_n \end{bmatrix} $，则$ T^{-1} H T = \begin{bmatrix} A - B R^{-1} B^T \hat{P} & -B R^{-1} B^T \\ 0 & -(A - B R^{-1} B^T \hat{P})^T \end{bmatrix} $，有$ \hat{P} = [\xi] [\eta]^{-1} $。
\item 解法：构造$ H = \begin{bmatrix} A & -B R^{-1} B^T \\ -Q & -A^T \end{bmatrix} $，求$ \lambda $和$ \begin{bmatrix} \eta \\ \xi \end{bmatrix} $，$ \hat{P} = [\xi] [\eta]^{-1} $。
\end{itemize}
\end{itemize}
%------------------------------------------------
\paragraph{定理}
\begin{itemize}
\item 方程与解：$ u = -R^{-1} B^T \hat{P} x $，其中$ \hat{P} $是以下方程的解。
$$ -\hat{P} A - A^T \hat{P} + \hat{P} B R^{-1} B^T \hat{P} - Q = 0 $$
\item 最优解：$ J^*[x(t_0), t_0]_{min} = \frac{1}{2} x^T(t_0) \hat{P} x(t_0) $。
\item 最优轨线$ \begin{cases} \frac{d x}{d t} = [A - B R^{-1} B^T \hat{P}] x(t) \\ x(t_0) = x_0 \end{cases} $。
\item 闭环系统$ \dot{x} = (A - B R^{-1} B^T \hat{P}) x $渐进稳定。
\end{itemize}

\subsubsection[有限时间时变输出跟踪系统]{\textcolor{red}{有限时间时变输出跟踪系统}}
使输出接近理想输出，\textcolor{red}{需已知任意时刻的$ y^d(t) $}。
\begin{align*}
&\begin{cases} \dot{x} = A(t) x + B(t) u \\ y = C(t) x \end{cases}, \quad x(t_0) = x_0 \\
&J(u) = \frac{1}{2} [y(t_f) - y^d(t_f)]^T F [y(t_f) - y^d(t_f)] + \frac{1}{2} \int_{t_0}^{t_f} \{[y(t) - y^d(t)]^T Q(t) [y(t) - y^d(t)] + u^T R u\} d t    
\end{align*}

其中$ Q $对称非负定，$ R $对称正定。

列哈密顿函数：
$$ H = \frac{1}{2}[C(t)x - y^d(t)]^T Q(t) [C(t)x - y^d(t)] + \frac{1}{2} u^T R u + \lambda^T [A(t) x + B(t) u] $$

得到：
\begin{align*} 
&\frac{\partial H}{\partial u} = 0 \Rightarrow u = -R^{-1}(t) B^T(t) \lambda(t) \\ 
&\dot{\lambda} = -\frac{\partial H}{\partial x} = -C^T(t) Q(t) [C(t) x - y^d(t)] - A^T(t) \lambda \\ 
&\dot{x} = A(t) x - B(t) R^{-1} B^T(t) \lambda \\
&\lambda(t_f) = \frac{\partial \theta[x(t_f), t_f]}{\partial x(t_f)} = \underbrace{C^T(t_f) F C(t_f)}_{P(t)} x(t_f) - \underbrace{C^T(t_f) F y^d(t_f)}_{\xi(t)}
\end{align*}

有：
\begin{align*}
&\begin{cases} \dot{P}(t) + P(t) A(t) + A^T(t) P(t) - P(t) B(t) R^{-1}(t) B^T(t) P(t) + C^T(t) Q(t) C(t) = 0 \\ P(t_f) = C^T(t_f) F C(t_f) \end{cases} \\
&\begin{cases} \dot{\xi}(t) = -[A(t) - B(t) R^{-1}(t) B^T(t) P(t)]^T \xi(t) - C^T(t) Q(t) y^d(t) \\ \xi(t_f) = C^T(t_f) F y^d(t_f) \end{cases}
\end{align*}

解得$ u^*(t) = -R^{-1}(t) B^T(t) [P(t) x(t) - \xi(t)] $，最优轨线为：
$$ \begin{cases} \dot{x}(t) = [A - B R^{-1} B^T P(t)] x(t) + B R^{-1} B^T \xi(t) \\ x(t_0) = x_0 \end{cases} $$

\subsubsection[带观测器的最优调节器]{\textcolor{red}{带观测器的最优调节器}}
$$ \begin{cases} \dot{x} = A x + B u \\ y = C x \\ \dot{z} = (A + GC)z + B u - G y \\ u = -k z + v, \quad k = R^{-1} B^T P \end{cases}, \quad J(u) = \frac{1}{2} \int_{t_0}^{\infty} (x^T Q x + u^T R u) d t $$

整理并进行变换：
$$
\begin{cases} \begin{bmatrix} \dot{x} \\ \dot{z} \end{bmatrix} = \begin{bmatrix} A & B k \\ G C & A + G C - B k \end{bmatrix} \begin{bmatrix} x \\ z \end{bmatrix} + \begin{bmatrix} B \\ B \end{bmatrix} u \\ y = \begin{bmatrix} C & 0 \end{bmatrix} \begin{bmatrix} x \\ z \end{bmatrix} \end{cases} 
\overset{\begin{bmatrix} x \\ z \end{bmatrix} = \begin{bmatrix} I_n & 0 \\ I_n & -I_n \end{bmatrix} \begin{bmatrix} x' \\ z' \end{bmatrix}}{\Longrightarrow}
\begin{cases} \overline{A} = \begin{bmatrix} A + B k & - B k \\ 0 & A + G C \end{bmatrix} \\ \overline{B} = \begin{bmatrix} B & 0 \end{bmatrix}^T \\ \overline{C} = \begin{bmatrix} C & 0 \end{bmatrix} \end{cases} 
$$
\begin{itemize}
\item $ \textcolor{red}{G(s)} $：$ \textcolor{red}{\overline{G}(s) = \overline{C}(s I - \overline{A})^{-1} \overline{B} = C[s I - (A - B K)]^{-1} B} $。
\item \textcolor{red}{极点}：\textcolor{red}{带观测器的最优调节器 = 观测器 + 最优调节器}。
\end{itemize}

\subsection[动态规划]{\textcolor{red}{动态规划}}
$ \dot{x} = f(x, u, t), \quad x(t_0) = x_0, \quad J(u) = \theta[x(t_f), t_f] + \int_{t_0}^{t_f} L(x, u, t) d t $。
%------------------------------------------------
\paragraph{HJB（连续系统动态规划基本方程）方程}
\begin{itemize}
\item $ -\frac{\partial J^*(x, t)}{\partial t} = \min_{u \in U} \{L(x, u, t) + [\frac{\partial J^*(x, t)}{\partial x}]^T f(x, u, t)\} $。
\item 边界条件：$ J^*[x(t_f), t_f] = \theta[x(t_f), t_f], \quad \lambda(t_f) = \frac{\partial \theta[x(t_f), t_f]}{\partial x(t_f)} $。
\item \textcolor{red}{哈密顿-雅可比方程}：$ -\frac{\partial J^*(x, t)}{\partial t} = \min_{u \in U} H(x, u, \lambda, t) $。
\item 协状态方程：$ \dot{\lambda}(t) = -\frac{\partial H(x, u, \lambda, t)}{\partial x} $。
\end{itemize}
%------------------------------------------------
\paragraph{解法}
\begin{enumerate}
\item $ H(x, u, \lambda, t) = L(x, u, t) + [\frac{\partial J^*(x, t)}{\partial x}]^T f(x, u, t) $。
\item 求$ u^* = g(x, \frac{\partial J^*(x, t)}{\partial x}, t) $：
$$ \begin{cases} \frac{\partial H}{\partial u} = 0, \quad& u \text{无约束} \\ \min H(x, u, \lambda, t), \quad& u \in U \end{cases} $$
\item $ u^* $代入以下方程组得到$ J^*[x(t), t] $，再代回$ u^*(t) $：
$$ \begin{cases} \text{哈密顿-雅可比方程：} -\frac{\partial J^*(x, t)}{\partial t} = \min_{u \in U} H(x, u, \lambda, t) \\ \text{边界条件：} J^*[x(t_f), t_f] = \theta[x(t_f), t_f] \end{cases} $$ 
\end{enumerate}

\subsection[最大值原理]{\textcolor{red}{最大值原理}}
\begin{align*} &x(k + 1) = f[x(k), u(k), k], \quad x(0) = x_0, \quad k = 0, 1, \dots, N - 1 \\ &J = \theta[x(N), N] + \sum_{k = 0}^{\textcolor{red}{N - 1}} L[x(k), u(k), k], \quad N \text{已知}, \quad x(N) \text{自由} \end{align*}
%------------------------------------------------
\paragraph{推导}
$ H(k) = L[x(k), u(k), k] + \lambda^T(k + 1) f[x(k), u(k), k] $。
\begin{itemize}
\item 
\begin{itemize}
\item $ \textcolor{red}{x(k + 1) = \frac{\partial H(k)}{\partial \lambda(k + 1)} = f[x(k), u(k), k]} $。
\item $ \textcolor{red}{\lambda(k) = \frac{\partial H(k)}{\partial x(k)}} \Rightarrow \lambda(k + 1) = m[\lambda(k), x(k)] $。
\end{itemize}
\item 边界条件：$ x(0) = x_0 $。
\item 横截条件：$ \textcolor{red}{\lambda(N) = \frac{\partial \theta[x(N), N]}{\partial x(N)}} $。
\item 极值目标：
\begin{itemize}
\item \textcolor{red}{$ u(k) $受约束}：$ \textcolor{red}{H^*(k) = \min_{u(k) \in U} H(k)} $。
\item \textcolor{red}{$ u(k) $不受约束}：$ \textcolor{red}{\frac{\partial H(k)}{\partial u(k)} = 0} \Rightarrow u(k) = n[\lambda(k + 1)] $。
\end{itemize}
\end{itemize}
%------------------------------------------------
\paragraph{\textcolor{red}{解法}（$ u(k) $不受约束）}\tip{最大值原理（离散）}
\begin{enumerate}
\item $ \lambda(N) = \frac{\partial \theta[x(N), N]}{\partial x(N)} = g_N[x(N)] $。
\item $ \lambda(k) = g_k[x(N)] \overset{x(k + 1) = f[x(k), u(k), k]}{\Longrightarrow} \lambda(k) = gf_k[x(k - 1), u(k - 1), k - 1] \overset{u(k) = n[\lambda(k + 1)]}{\Longrightarrow} \lambda(k) = gf_k'[x(k - 1), k - 1] \overset{\lambda(k + 1) = m[\lambda(k), x(k)]}{\Longrightarrow} \lambda(k - 1) = g_{k - 1}[x(k - 1)] $，迭代到$ \lambda(1) = gf_{1}[x(0)] $。
\item 逐代回归表示$ u(k), \lambda(k), x(k) $为关于$ x(0) $的函数。
\end{enumerate}
%------------------------------------------------
\subsection[线性二次型]{\textcolor{red}{线性二次型}}
\begin{align*} &x(k + 1) = A(k) x(k) + B(k) u(k), \quad x(0), \quad k = 0, 1, \dots, N - 1 \\ &J = \frac{1}{2} x^T(x) \textcolor{red}{F} x(N) + \frac{1}{2} \sum_{k = 0}^{N - 1} [x^T(k) \textcolor{red}{Q(k)} x(k) + u^T(k) \textcolor{red}{R(k)} u(K)] \\ &N \text{已知}, \quad x(N) \text{自由}, \quad Q(k) \text{对称半正定}, \quad R(k) \text{对称正定} \end{align*}
%------------------------------------------------
\paragraph{推导}
$$ H(k) = \frac{1}{2}[x^T(k) Q(k) x(k) + u^T(k) R(k) u(K)] + \lambda^T(k + 1)[A(k) x(k) + B(k) u(k)] $$
\begin{itemize}
\item $ \lambda(k) = \frac{\partial H(k)}{\partial x(k)} = Q(k) x(k) + A^T(k) \lambda(k + 1) $。
\item 横截条件：$ \lambda(N) = F x(N) $。
\item 极值目标：$ \frac{\partial H(k)}{\partial u(k)} = 0 \Rightarrow u(k) = -R^{-1}(k) B^T(k) \lambda(k + 1) $。
\end{itemize}
%------------------------------------------------
\paragraph{定理}
$ u^*(k) = -[R(k) + B^T(k) P(k + 1) B(K)]^{-1} B^T(k) P(k + 1) A(k) x(k) $得到$ J_{min}^* = \frac{1}{2} x^T(0) P(0) x(0) $，$ N \to \infty $时$ P(N) $为常阵。
%------------------------------------------------
\paragraph{解法}
\begin{itemize}
\item 法1：$ Z $迭代
\begin{enumerate}
\item 因为$ \lambda(k) = P(k) x(k), \quad \lambda(N) = F x(N)  $，故$ P(N) = F $。
\item 对$ k = N - 1, N - 2, \dots, 0 $，求$ Z_1(k), Z_2(k), Z_3(k) $，进而求$ P(K), u^*(k) $。
\begin{align*} Z_1(k) &= Q(k) + A^T(k) P(k + 1) A(k) \\ Z_2(k) &= B^T(k) P(k + 1) A(k) \\ Z_3(k) &= R(K) + B^T(k) P(k + 1) B(k) \\ P(k) &= Z_1(k) - Z_2^T(k) Z_3^{-1}(k) Z_2(k) \\ u^*(k) &= -Z_3^{-1}(k) Z_2(k) x(k) \end{align*}
\end{enumerate}
\item \textcolor{red}{法2}\tip{线性二次型（离散）}：$ P $迭代
\begin{enumerate}
\item \textcolor{red}{任取对称正定阵$ P_0 $}（如$ I $）。
\item 求\textcolor{red}{$ P(k) $与$ P(k + 1) $间的递推关系}：
$$ \textcolor{red}{P(k) = Q + A^T P(k + 1) A - A^T P(k + 1) B [R + B^T P(k + 1) B]^{-1} B^T P(k + 1) A} $$
\item 代$ P(k + 1) = P_0 $，得$ P(k) = P_1 $；$ \dots $；代$ P(k + 1) = P_i $，得$ P(k) = P_{i + 1} $。
\item 当\textcolor{red}{$ P_i = P_{i + 1} $时，停止迭代}，$ \textcolor{red}{u^*(k) = -[R + B^T P_i B]^{-1} B^T P_i A x(k)} $。
\end{enumerate}
\end{itemize}

\subsection[动态规划]{\textcolor{red}{动态规划}\tip{动态规划原理}}
%------------------------------------------------
\paragraph{概念}
$$ \begin{cases} x(k + 1) = f[x(k), u(k), k] \\ J_N = \sum_{k = 0}^{N - 1} L[x(k), u(k), k] + \theta[x(N)] \end{cases} $$
\begin{itemize}
\item 要求
\begin{itemize}
\item 无后效性。
\item 目标函数可分性。
\end{itemize}
\item 将状态方程逐阶代入目标函数，可得$ J_N = J_N[x(0), u(0), u(1), \dots, u(N - 1)] $。
\item \textcolor{red}{最优性原理}：$ u^*(0), \dots, u^*(N - 1) $最优，则$ u^*(k), \dots, u^*(N - 1) $最优。
\end{itemize}
%------------------------------------------------
\paragraph{\textcolor{red}{动态规划基本方程}}
$$ \begin{cases} J_{N - i}^*[x(i)] = \min_{u(i)} \{L[x(i), u(i), i] + J_{N - i - 1}^*[x(i + 1)]\} \\ x(i + 1) = f[x(i), u(i)] \end{cases} $$
%------------------------------------------------
\paragraph{解法}\tip{动态规划（离散）}
\begin{enumerate}
\item $ J_0[x(N)] = \theta[x(N)] $。
\item $ \textcolor{red}{J_i[x(N - i)] = J_{N - i - 1}^*[x(i + 1)] + L[x(i), u(i), i]} $，代入$ x(i + 1) = f[x(i), u(i)] $，\textcolor{red}{求最值}（对$ u_i $求偏导），得到$ u(i) $关于$ x(i) $的函数，迭代到$ J_N[x(0)] $。
\item 逐代回归表示$ u(k), x(k) $为关于$ x(0) $的函数。
\end{enumerate}
